# fa_test (<program> <kinds> [WILL_FAIL]): one ctest per allocator kind, on CPU, and on GPU if enabled
set (SIDES cpu)
if (ENABLE_GPU)
  list (APPEND SIDES gpu)
endif ()
function (fa_test PROG KINDS)
  foreach (S ${SIDES})
    foreach (K ${KINDS})
      add_test (NAME ${PROG}_${K}_${S} COMMAND ${PROG} ${K} ${S})
      if (ARGN STREQUAL "WILL_FAIL")
        set_tests_properties (${PROG}_${K}_${S} PROPERTIES WILL_FAIL TRUE)
      endif ()
    endforeach ()
  endforeach ()
endfunction ()

fa_test (test_types          "doublestack;fixed;peak")
fa_test (test_sides          "doublestack;fixed;peak")
fa_test (test_addresses      "doublestack;fixed")
fa_test (test_peak           "doublestack;fixed;peak")
fa_test (test_debug          "doublestack;fixed")
fa_test (test_lifo_violation "doublestack;fixed" WILL_FAIL)

# GPU-only tests: offload kernels (OpenACC or OpenMP); graphs (NVIDIA: OpenACC kernels in a CUDA
# graph; AMD: hipBLAS SGEMMs in a HIP graph, since OpenMP target regions cannot be captured);
# layout of the HIP structures (AMD)
if (ENABLE_GPU)
  if (GPU_VENDOR STREQUAL "nvidia")
    set (GRAPH_SOURCE cuda_graph_mod.F90)
  else ()
    set (GRAPH_SOURCE hip_graph_mod.F90)
    get_filename_component (HIP_LIBDIR ${HIP_LIBRARY} DIRECTORY)
    find_library (HIPBLAS_LIBRARY hipblas HINTS ${HIP_LIBDIR} REQUIRED)
    target_link_libraries (test_util PUBLIC ${HIPBLAS_LIBRARY})
  endif ()
  add_executable (test_offload test_offload.F90)
  target_link_libraries (test_offload test_util)
  add_executable (test_graph ${GRAPH_SOURCE} test_graph.F90)
  target_link_libraries (test_graph test_util)
  foreach (K doublestack fixed peak)
    add_test (NAME test_offload_${K}_gpu COMMAND test_offload ${K} gpu)
    add_test (NAME test_graph_${K}_gpu COMMAND test_graph ${K} gpu)
  endforeach ()
  if (GPU_VENDOR STREQUAL "amd")
    enable_language (C)
