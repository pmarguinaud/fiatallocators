# fa_test (<program> <kinds> [WILL_FAIL]): one ctest per allocator kind, on CPU, and on GPU if enabled
set (SIDES cpu)
if (ENABLE_GPU)
  list (APPEND SIDES gpu)
endif ()
function (fa_test PROG KINDS)
  foreach (S ${SIDES})
    foreach (K ${KINDS})
      add_test (NAME ${PROG}_${K}_${S} COMMAND ${PROG} ${K} ${S})
      if (ARGN STREQUAL "WILL_FAIL")
        set_tests_properties (${PROG}_${K}_${S} PROPERTIES WILL_FAIL TRUE)
      endif ()
    endforeach ()
  endforeach ()
endfunction ()

fa_test (test_types          "doublestack;fixed;peak")
fa_test (test_sides          "doublestack;fixed;peak")
fa_test (test_addresses      "doublestack;fixed")
fa_test (test_peak           "doublestack;fixed;peak")
fa_test (test_debug          "doublestack;fixed")
fa_test (test_lifo_violation "doublestack;fixed" WILL_FAIL)
fa_test (test_fixed_full     "fixed" WILL_FAIL)
